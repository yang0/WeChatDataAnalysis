cmake_minimum_required(VERSION 3.20)
project(magpie_wechat_key LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 build required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
if(NOT DEFINED UPSTREAM)
  message(FATAL_ERROR "Pass pinned py_wx_key source with -DUPSTREAM=...")
endif()
set(U "${UPSTREAM}/py_wx_key")
add_library(wx_key SHARED "${U}/src/hook_controller.cpp" "${U}/src/ipc_manager.cpp" "${U}/src/remote_hooker.cpp" "${U}/src/remote_scanner.cpp" "${U}/src/remote_veh.cpp" "${U}/src/shellcode_builder.cpp" "${U}/src/syscalls.cpp" "${U}/src/veh_hook_manager.cpp")
target_include_directories(wx_key PRIVATE "${U}/include" "${U}/vendor/MinHook_134_lib/include" "${U}/vendor/xbyak")
target_compile_definitions(wx_key PRIVATE HOOK_EXPORTS)
target_compile_options(wx_key PRIVATE /utf-8)
target_link_libraries(wx_key PRIVATE "${U}/vendor/MinHook_134_lib/lib/libMinHook.x64.lib" Psapi version Crypt32 Advapi32 Shell32 Ole32)
add_library(fake_key SHARED fake-key.cpp)
